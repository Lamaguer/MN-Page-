            fa <-fx
        }

Método de Laguerre

Calcula raíces de polinomios, incluyendo raíces complejas.

#Metodo de Laguerre
coef <- c(1,0,1) #Es la formula (x2+1, pero solo en coeficientes.)
x0 <-1 #Punto de partida
iter <- 0
maxi<-100
tol <- 1e-6

n <- length(coef)-1  #Grado polinomial
x <- as.complex(x0) #Complex añade +0i para que no genere error al dar imaginario

while(iter<maxi){
    iter<-iter+1

    P <- as.complex(coef[1])
    dP <- as.complex(0)
    ddP <- as.complex(0)

    for(i in 2:(n+1)){ #Hacerlo al reves previene generar variables temporales
        ddP <- ddP * x +2 *dP
        dP <- dP*x + P
        P <- P*x + coef[i]
    }

    if(Mod(P)< tol){    #Se usa Mod para tomar en cuenta el numero imaginario
        break
    }

    G <- dP/ P
    H <- G^2 - (ddP/P)
    radicando<- (n-1)*(n*H -G^2)
    raiz_rad <- sqrt(radicando)

    den1 <- G+ raiz_rad
    den2 <- G- raiz_rad
    denominador <- if(Mod(den1)>= Mod(den2)){
        den1
    }else{ 
        den2
    }
    paso <- n/denominador
    x <- x - paso
    #Se compara paso ya que paso no es un numero complejo (i)
    if(Mod(paso) < tol){
        break
    }
}
    cat("Iteraciones totales: ",iter, "\n")

Factorización LDLᵀ

Resuelve un sistema lineal con una matriz simétrica.

#Factorizacion LDL T 

A <- matrix(c(4,1,
              1,3),nrow=2, byrow=TRUE)
B <- c(1,2)

n<-nrow(A)
if(n!=ncol(A) || !isSymmetric(A)){
    stop("Matriz A tiene que ser cuadrada y simétrica.")
}

L <- diag(1,n) #Crea una matriz de tamaño n con 1 en la diagonal y 0 en todo lo demoas
D <- numeric(n) #Crea un vector de n tamaño vacio.

for(j in 1:n){
    #Diagonal D
    suma_d <- 0
    if(j >1){
     for(k in 1:(j-1)){ #D se construye usando la sumatoria
        suma_d <- suma_d + (L[j,k]^2)*D[k] 
     }   
    }
    D[j] <- A[j,j] - suma_d
    if(D[j]==0){
        stop("Pivote nulo encontrado en D[",j,"]. No se puede tener pivote nulo en este metodo.")
    }
    #Elementos inferiores L
    if(j<n){
        for(i in (j+1):n){
            sumal <- 0
            if(j> 1){
                for(k in 1:(j-1)){
                    sumal<- sumal + L[i,k]*L[j,k] * D[k]
                }
            }
            L[i,j] <- (A[i,j]- sumal)/ D[j]
        }
    }
}
#Lz =B
z <- numeric(n)
for(i in 1:n){
    suma <-0
    if(i >1){
        for(j in 1:(i-1)){
            suma <- suma + L[i,j]*z[j]
        }
    }
    z[i] <- B[i]-suma
}   
#Dw = z
w<- z/D

#LTx = w
x<-numeric(n)
for(i in n:1){
    suma <-0
    if(i <n){
        for(j in (i+1):n){
            suma <- suma + L[j,i]*x[j]
        }
    }
    x[i] <- w[i] - suma
}

Reducción cíclica

Resuelve un sistema tridiagonal mediante reducción de ecuaciones.

#Reduccion ciclica

# 2x1 -x2 + 0x3 = 1
# -x1 +2x2 -x3 = 0
# 0x1+ -x2 + 2x3 = 1

a <- c(0,-1,-1)
d <- c(2,2,2)
c <- c(-1,-1,0)
b <- c(1,0,1)
n<- length(d)
if(length(a)!= n || length(b) !=n || length(c) !=n ){
    stop("Los vectores necesitan ser del a,d,c y b deben de tener el mismo tamaño que n.")
}
#Estos valores deben de ser 0 porque el metodo necesita un sistema tridiagonal
a[1] <- 0
c[n] <- 0

#inicializacion
d_red <- numeric(n)
b_red <- numeric(n)
a_red <- numeric(n)
c_red <- numeric(n)

for (i in seq(1,n, by=2)){
    if(i-1>=1){ #Checa si hay vecino izquierda
        alpha <- -a[i]/d[i-1]
        c_prev <- c[i-1]
        b_prev <- b[i-1]
        a_prev <- a[i-1]
    }else{
        alpha <-  0
        c_prev <- 0
        b_prev <- 0
        a_prev <- 0
    }
    if(i+1 <=n){ #NCheca si no hay vecino derecho
        beta <- -c[i] / d[i+1]
        a_next <- a[i+1]
        b_next <- b[i+1]
        c_next <- c[i+1]
    }else{
        beta <- 0
        a_next <- 0
        b_next <- 0
        c_next <- 0
    }
    #red de reducida 
    d_red[i] <- d[i] + alpha * c_prev + beta * a_next
    b_red[i] <- b[i] + alpha * b_prev + beta * b_next
    a_red[i] <- alpha * a_prev
    c_red[i] <- beta * c_next
}

x <- numeric(n)
#Regla de Cramer
det <- d_red[1] * d_red[3] - c_red[1] * a_red[3]
x[1] <- (b_red[1]*d_red[3] - c_red[1] * b_red[3])/det
x[3] <- (d_red[1]*b_red[3] - b_red[1] * a_red[3])/det

#Reconstruccion regresiva
for(i in seq(2,n,by=2)){
    x[i] <- (b[i]-a[i] * x[i-1] - c[i]*x[i+1]) / d[i]
}

cat("Solucion x:", sprintf("%.4f",x ),"\n")
Solucion x: 1.0000 1.0000 1.0000

Clenshaw-Curtis

Calcula una integral usando nodos y pesos de Chebyshev.

#Metodo Clenshaw-Curtis
#Funcion f(x):
# n=5
fx <- function(x){ return(x^4)}
a<- -1 #Limite inferior
b<- 1 #Limite superior
n <- 5

if (n<3){
    stop("Clenshaw Curtis requiere n=3 nodos como minimo.")
}

N <- n-1
j <- 0:N

x <- cos(j*pi/ N)
w <- numeric(n)
kmax <- floor((N-1)/2)

for (i in 1:(n)){
    j_val <- j[i]
    #Si es esquina es 0.5, si no 1.
    deltaj <- if(j_val ==0 || j_val==N) 0.5 else 1.0 

    suma <- 0
    if(kmax >= 1 ){
        for(k in 1:kmax){
            suma <- suma + (2*cos(2*k * j_val * pi / N)) / (4*k^2 - 1)
        }
    }
    Ej <- if(N %% 2 ==0) cos(j_val * pi)/(N^2 - 1) else 0

    w[i] <- (2*deltaj / N) * (1 - suma - Ej)
}
#Transformacion de nodos al intervalo a,b
t <- ((a+b)/2) + ((b-a)/2)*x

res <- ((b-a)/2) * sum(w* fx(t))

cat("Resultado del metodo con n=",n,": ",sprintf("%.8f",res),"\n")
Resultado del metodo con n= 5 :  0.40000000

# y' = 5yt - 1
f <- function(t,y){return(5*y*t - 1)
}

y0 <- 2
n <- 10
if(n <2){
    stop("Se requieren al menos n=2 pasos")
}

h <- (tf-t0)/n
t <- seq(t0,tf, by=h)
y <- numeric(n+1)

y[1] <- y0

#RK4 Para obtener y1 en t1
k1 <- f(t[1], y[1])
k2 <- f(t[1] + h/2, y[1]+ (h/2) * k1)
k3 <- f(t[1] + h/2, y[1]+ (h/2) * k2)
k4 <- f(t[1] + h, y[1] + h * k3)

y[2] <- y[1] + (h/6) * (k1 * 2 * k2 * 2 * k3 + k4)

# Adam Bashforth
for( k in 2:n){
    fk <- f(t[k], y[k])
    fkant <- f(t[k-1], y[k-1])

    y[k+1] <- y[k] + (h/2) * (3 * fk - fkant)
}

Predictor-corrector de Hamming

Resuelve una ecuación diferencial ordinaria mediante un método predictor-corrector modificado.

# Predictor corrector de Hamming modificado
# y1, y2 y 3 obtenidas por Runge-Kutta 4to Orden.

# y' = 5yt - 1
f <- function(t,y){return(5*y*t - 1)}
t0 <- 0
tf <- 1
y0 <- 2
n <- 10

if(n<4){
    stop("Hamming modificado requiere al menos n=4 pasos")
}

h <-(tf-t0)/n
t <- seq(t0,tf,by =h)
y <- numeric(n+1)
y[1] <- y0

#Runge-Kutta 4t0 Orden
for(k in 1:3){
    tk <- t[k]
    yk <- y[k]

    k1 <- f(tk, yk)
    k2 <- f(tk+h / 2, yk + (h/2) * k1)
    k3 <- f(tk+h / 2, yk + (h/2) * k2)
    k4 <- f(tk + h, yk + h * k3)
    y[k+1] <- yk + (h/6) * (k1 +  2*k2 + 2*k3 + k4)
}

p <- numeric(n+1)
c <- numeric(n+1)
#La diferencia previa inicial siempre inicia en 0 
#al ser el primer recorrimiento del metodo.
diff_prev <- 0 

#Predictor corrector de Hamming modificado
for(k in 4:n){
    fk <- f(t[k],y[k])
    fkm1 <- f(t[k-1],y[k-1])
    fkm2 <- f(t[k-2],y[k-2])
    
    #Predictor
    pk <- y[k-3] + (4*h /3) * (2*fk - fkm1 + 2*fkm2)
    #Predictor modificado
    mk <- pk + (112/121) * diff_prev
    #Corrector
    fm <- f(t[k+1], mk)
    ck <- (9 * y[k] - y[k-2])/ 8 + (3*h / 8) * (2*fk - fkm1 + fm)

    y[k+1] <- ck + (9/121) * (pk - ck)

    #Actualizar para el siguiente ciclo.
    p[k+1] <- pk
    c[k+1] <- ck
    diff_prev <- ck - pk
}

res <- data.frame(t=t, y=y)
cat("Resultado de los pasos hasta t = 1.0: \n")

Interpolación de Shepard

Este script solicita puntos y parámetros con readline(). Por esa razón se muestra aquí sin ejecutarse automáticamente durante la generación.

# Metodo de Interpolación de Shepard

j <- as.numeric(readline(prompt="Ingrese cantidad de puntos a interpolar"))
print("Ingrese las coordenadas del punto de consulta:")
x_consulta <- as.numeric(readline(prompt="Eje X"))
y_consulta <- as.numeric(readline(prompt="Eje Y"))
p <- as.numeric(readline(prompt="Exponente (P)"))

x_pts <- numeric(j)
y_pts <- numeric(j)
z_pts <- numeric(j)

for(i in 1:j){
    cat("Punto [",i,"]")
    x_pts[i] <- as.numeric(readline(prompt= paste("X del punto [",i,"]")))
    y_pts[i] <- as.numeric(readline(prompt= paste("Y del punto [",i,"]")))
    z_pts[i] <- as.numeric(readline(prompt= paste("Z del punto [",i,"]")))
}

dists <- sqrt((x_consulta-x_pts)^2 + (y_consulta-y_pts)^2)

coincidencia <- which(dists==0)
if (length(coincidencia) >0){
    return(z_pts[coincidencia])
}

pesos <- 1/(dists^p)

Sz <- sum(pesos*z_pts)/sum(pesos)

cat("El valor interpolado de todos los puntos es: ", Sz)

Spline cúbico natural

Este script solicita los datos mediante readline(). Se muestra aquí sin ejecutarse automáticamente durante la generación.

z<-0
dats<-0
z<- as.numeric(readline(prompt="Ingrese el valor de z: "))
dats<- as.numeric(readline(prompt="Ingrese la cantidad de datos a interpolar: "))
if(dats<3){
    cat("ERROR, El número de datos debe ser mayor o igual a 3 para interpolar con este metodo.")
}else{
    vectorx <- vector("numeric", dats)
    vectory <- vector("numeric", dats)
    for(i in 1:dats){
        vectorx[i] <- as.numeric(readline(prompt = paste("Ingrese el valor de x",i, ": ")))
        vectory[i] <- as.numeric(readline(prompt = paste("Ingrese el valor de y",i, ": ")))
    }
    h <- diff(vectorx)
    d <- diff(vectory) / h
    n_int <- dats-2
    A <- matrix(0, nrow = n_int, ncol = n_int)
    B <- numeric(n_int)

    for(i in 1:n_int){
        A[i,i] <- 2 * (h[i]+h[i+1])
        if(i > 1) A[i, i-1 ] <- h[i]
        if(i < n_int) A[i, i+1] <- h[i+1]

        B[i] <- 6 * (d[i+1] - d[i])
    }

    M_int <- solve(A,B)
    M <- c(0, M_int, 0)

    #En que subintervalo se encuentra z
    i <- max(which(vectorx <= z))
    if(i==dats){
        i <- dats-1
    }

    h_i <- vectorx[i+1] - vectorx[i]
    A_rel <- (vectorx[i+1]- z)/h_i
    B_rel <- (z - vectorx[i])/h_i

    Sz <- A_rel*vectory[i] + B_rel*vectory[i+1] + (h_i^2 / 6)*((A_rel^3 - A_rel)* M[i] + (B_rel^3 - B_rel)*M[i+1])
    
    cat("Valor interpolado de Z:", Sz)
}
